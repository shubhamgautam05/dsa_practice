class Solution {
public:
    int largestAltitude(vector<int>& gain) {
        vector<long long> v(gain.size()+1);
        long long sum=0;
        for(int i=0;i<gain.size();i++){
            v[i+1]=gain[i]+sum;
            sum+=gain[i];
        }
        int max=*max_element(v.begin(),v.end());
        return max;
    }
};
