class Solution {
public:
    string frequencySort(string s) {
    unordered_map<char,int> freq;
    for (char c : s) freq[c]++;
    vector<pair<int,char>> arr;
    for (auto p : freq) arr.push_back({p.second, p.first});
    sort(arr.rbegin(), arr.rend());            
    string ans;
    for (auto p : arr) ans += string(p.first, p.second);
    return ans;
}
};
