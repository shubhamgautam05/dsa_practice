//50. Pow(x, n)
class Solution {
public:
    double f(double x, long long n) {
        if (n == 0) return 1;
        double t = f(x, n / 2);
        if (n % 2 == 0)
            return t * t;
        else
            return t * t * x;
    }
    double myPow(double x, int n) {
        long long p = n;
        if (p < 0) {
            x = 1 / x;
            p = -p;
        }
        return f(x, p);
    }
};
