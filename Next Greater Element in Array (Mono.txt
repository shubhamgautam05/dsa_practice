Next Greater Element in Array (Monotonic Stack)
class Solution {
  public:
    vector<int> nextLargerElement(vector<int>& a) {
        int n = a.size();
        vector<int> v(n);
        stack<int> s;

        for(int i = n - 1; i >= 0; i--) {
            while(!s.empty() && s.top() <= a[i]) {
                s.pop();
            }

            if(s.empty()) v[i] = -1;
            else v[i] = s.top();

            s.push(a[i]);
        }

        return v;
    }
};
