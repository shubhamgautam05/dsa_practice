class Solution {
  public:
    Node *insertAtEnd(Node *head, int x) {
        Node* newNode = new Node(x);
        if (head==NULL) {
            return newNode;
        }
        Node* ptr=head;
        while (ptr->next != NULL) {
            ptr = ptr->next;
        }
        ptr->next=newNode;
        return head;
    }
};
