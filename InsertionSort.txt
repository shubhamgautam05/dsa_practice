#include <bits/stdc++.h>
using namespace std;
void insertionSort(vector<int>& arr) {
    int n = arr.size();
    for (int i = 1; i < n; i++) {
        int key = arr[i];
        int j = i - 1;
        while (j >= 0 && arr[j] > key) {
            arr[j + 1] = arr[j];
            j--;
        }
        arr[j + 1] = key;
    }
    cout<<"Insertion Sort: ";
    for (int x : arr) {
        cout << x << " ";
    }
    cout<<endl;
}
int main() {
    // Write C++ code here
    vector<int> v={8,45,203,5,23,4,25,27,2,0,86,15,1,9};
    insertionSort(v);

    return 0;
}
